   410→   410→export function getProductsForRegion(region: string, budget?: string, skinTypes?: string[], hairType?: string[]): RegionalProduct[] {
   411→   411→  let products = regionalProducts.filter((p) => p.region === region);
   412→   412→
   413→   413→  // If no products for exact region, show all as fallback
   414→   414→  if (products.length === 0) {
   415→   415→    products = [...regionalProducts];
   416→   416→  }
   417→   417→
   418→   418→  // Filter by budget
   419→   419→  if (budget) {
   420→   420→    const budgetOrder = ['economico', 'moderado', 'premium', 'luxo'];
   421→   421→    const userBudgetIdx = budgetOrder.indexOf(budget);
   422→   422→    if (userBudgetIdx >= 0) {
   423→   423→      products = products.filter((p) => {
   424→   424→        const pIdx = budgetOrder.indexOf(p.budgetLevel);
   425→   425→        return pIdx <= userBudgetIdx + 1;
   426→   426→      });
   427→   427→    }
   428→   428→  }
   429→   429→
   430→   430→  // Filter by skin type compatibility
   431→   431→  if (skinTypes && skinTypes.length > 0) {
   432→   432→    const skinFiltered = products.filter((p) => {
   433→   433→      if (!p.skinType || p.skinType.length === 0) return true;
   434→   434→      return p.skinType.some((st) => skinTypes.includes(st));
   435→   435→    });
   436→   436→    if (skinFiltered.length > 0) products = skinFiltered;
   437→   437→  }
   438→   438→
   439→   439→  // Filter by hair type compatibility
   440→   440→  if (hairType) {
   441→   441→    const hairFiltered = products.filter((p) => {
   442→   442→      if (!p.hairType || p.hairType.length === 0) return true;
   443→   443→      return p.hairType.includes(hairType);
   444→   444→    });
   445→   445→    if (hairFiltered.length > 0) products = hairFiltered;
   446→   446→  }
   447→   447→
   448→   448→  return products.sort((a, b) => b.rating - a.rating);
   449→   449→}
